    85→/* ─── Helpers ─── */
    86→function formatDate(dateStr: string | null): string {
    87→  if (!dateStr) return '—'
    88→  return new Date(dateStr).toLocaleDateString('en-NG', {
    89→    month: 'short',
    90→    day: 'numeric',
    91→    hour: '2-digit',
    92→    minute: '2-digit',
    93→  })
    94→}

   123→/* ─── Main Component ─── */
   124→interface EmailDashboardProps {
   125→  user: any
   126→  token: string | null
   127→}
   128→
   129→export function EmailDashboard({ user, token }: EmailDashboardProps) {
   130→  /* ─── State ─── */
   131→  const [subTab, setSubTab] = React.useState<'compose' | 'broadcast' | 'history'>('compose')
   132→  const [stats, setStats] = React.useState<EmailStats>({ total: 0, sent: 0, failed: 0, pending: 0, recentSent: 0 })
   133→  const [isLoadingStats, setIsLoadingStats] = React.useState(true)
   134→
   135→  // History
   136→  const [emails, setEmails] = React.useState<EmailRow[]>([])
   137→  const [isLoadingEmails, setIsLoadingEmails] = React.useState(false)
   138→  const [historyPage, setHistoryPage] = React.useState(1)
   139→  const [historyTotal, setHistoryTotal] = React.useState(0)
   140→  const [statusFilter, setStatusFilter] = React.useState('ALL')
   141→  const [searchQuery, setSearchQuery] = React.useState('')
   142→
   143→  // Compose email
   144→  const [isSending, setIsSending] = React.useState(false)
   145→  const [recipientSearch, setRecipientSearch] = React.useState('')
   146→  const [recipientResults, setRecipientResults] = React.useState<any[]>([])
   147→  const [selectedRecipient, setSelectedRecipient] = React.useState<any>(null)
   148→  const [composeSubject, setComposeSubject] = React.useState('')
   149→  const [composeTemplate, setComposeTemplate] = React.useState('custom')
   150→  const [composeMessage, setComposeMessage] = React.useState('')
   151→  const [composeCtaUrl, setComposeCtaUrl] = React.useState('')
   152→  const [composeCtaLabel, setComposeCtaLabel] = React.useState('')
   153→
   154→  // Broadcast
   155→  const [isBroadcasting, setIsBroadcasting] = React.useState(false)
   156→  const [broadcastRole, setBroadcastRole] = React.useState('ALL')
   157→  const [broadcastSubject, setBroadcastSubject] = React.useState('')
   158→  const [broadcastTemplate, setBroadcastTemplate] = React.useState('announcement')
   159→  const [broadcastMessage, setBroadcastMessage] = React.useState('')
   160→
   161→  /* ─── Auth headers helper ─── */
   162→  const getHeaders = () => ({
   163→    'Content-Type': 'application/json',
   164→    ...(token ? { Authorization: `Bearer ${token}` } : {}),
   165→  })
   166→
   167→  /* ─── Fetch Stats ─── */
   168→  const fetchStats = React.useCallback(async () => {
   169→    setIsLoadingStats(true)
   170→    try {
   171→      const res = await fetch('/api/email/stats', { headers: getHeaders() })
   172→      const data = await res.json()
   173→      if (res.ok) setStats(data)
   174→    } catch (error) {
   175→      console.error('Error fetching email stats:', error)
   176→    } finally {
   177→      setIsLoadingStats(false)
   178→    }
   179→  }, [token])
   180→
   181→  /* ─── Fetch History ─── */
   182→  const fetchHistory = React.useCallback(async () => {
   183→    setIsLoadingEmails(true)
   184→    try {
   185→      const params = new URLSearchParams()
   186→      params.set('page', String(historyPage))
   187→      if (statusFilter && statusFilter !== 'ALL') params.set('status', statusFilter)
   188→      if (searchQuery) params.set('search', searchQuery)
   189→
   190→      const res = await fetch(`/api/email/history?${params.toString()}`, { headers: getHeaders() })
   191→      const data = await res.json()
   192→      if (res.ok) {
   193→        setEmails(data.emails || [])
   194→        setHistoryTotal(data.pagination?.total || 0)
   195→      }
   196→    } catch (error) {
   197→      console.error('Error fetching email history:', error)
   198→    } finally {
   199→      setIsLoadingEmails(false)
   200→    }
   201→  }, [historyPage, statusFilter, searchQuery, token])
   202→
   203→  /* ─── Search Recipients ─── */
   204→  const searchRecipients = React.useCallback(async (query: string) => {
   205→    if (!query || query.length < 2) {
   206→      setRecipientResults([])
   207→      return
   208→    }
   209→    try {
   210→      const params = new URLSearchParams()
   211→      params.set('search', query)
   212→      params.set('limit', '10')
   213→      const res = await fetch(`/api/admin/users?${params.toString()}`, { headers: getHeaders() })
   214→      const data = await res.json()
   215→      if (res.ok) setRecipientResults(data.users || [])
   216→    } catch (error) {
   217→      console.error('Error searching recipients:', error)
   218→    }
   219→  }, [token])
   220→
   221→  React.useEffect(() => {
   222→    fetchStats()
   223→  }, [fetchStats])
   224→
   225→  React.useEffect(() => {
   226→    if (subTab === 'history') fetchHistory()
   227→  }, [subTab, fetchHistory])
   228→
   229→  React.useEffect(() => {
   230→    const timer = setTimeout(() => searchRecipients(recipientSearch), 300)
   231→    return () => clearTimeout(timer)
   232→  }, [recipientSearch, searchRecipients])
   233→
   234→  /* ─── Send Single Email ─── */
   235→  const handleSendEmail = async () => {
   236→    if (!selectedRecipient || !composeSubject) {
   237→      toast.error('Please select a recipient and enter a subject')
   238→      return
   239→    }
   240→    setIsSending(true)
   241→    try {
   242→      const res = await fetch('/api/email/send', {
   243→        method: 'POST',
   244→        headers: getHeaders(),
   245→        body: JSON.stringify({
   246→          recipientId: selectedRecipient.id,
   247→          subject: composeSubject,
   248→          templateId: composeTemplate,
   249→          message: composeMessage,
   250→          ctaUrl: composeCtaUrl || undefined,
   251→          ctaLabel: composeCtaLabel || undefined,
   252→        }),
   253→      })
   254→      const data = await res.json()
   255→
   256→      if (!res.ok) {
   257→        toast.error(data.error || 'Failed to send email')
   258→        return
   259→      }
   260→
   261→      toast.success(data.message || 'Email sent successfully')
   262→      // Reset form
   263→      setSelectedRecipient(null)
   264→      setRecipientSearch('')
   265→      setComposeSubject('')
   266→      setComposeMessage('')
   267→      setComposeCtaUrl('')
   268→      setComposeCtaLabel('')
   269→      fetchStats()
   270→    } catch (error) {
   271→      console.error('Send email error:', error)
   272→      toast.error('Failed to send email')
   273→    } finally {
   274→      setIsSending(false)
   275→    }
   276→  }
   277→
   278→  /* ─── Broadcast Email ─── */
   279→  const handleBroadcast = async () => {
   280→    if (!broadcastSubject) {
   281→      toast.error('Please enter a subject')
   282→      return
   283→    }
   284→    setIsBroadcasting(true)
   285→    try {
   286→      const body: any = {
   287→        subject: broadcastSubject,
   288→        templateId: broadcastTemplate,
   289→        message: broadcastMessage,
   290→      }
   291→
   292→      if (broadcastRole && broadcastRole !== 'ALL') {
   293→        body.roleFilter = broadcastRole
   294→      }
   295→
   296→      const res = await fetch('/api/email/broadcast', {
   297→        method: 'POST',
   298→        headers: getHeaders(),
   299→        body: JSON.stringify(body),
   300→      })
   301→      const data = await res.json()
   302→
   303→      if (!res.ok) {
   304→        toast.error(data.error || 'Failed to broadcast')
   305→        return
   306→      }
   307→
   308→      toast.success(data.message || `Broadcast sent to ${data.totalSent} users`)
   309→      setBroadcastSubject('')
   310→      setBroadcastMessage('')
   311→      fetchStats()
   312→    } catch (error) {
   313→      console.error('Broadcast error:', error)
   314→      toast.error('Failed to broadcast email')
   315→    } finally {
   316→      setIsBroadcasting(false)
   317→    }
   318→  }
   319→
   320→  /* ─── Render ─── */
   321→  return (
   322→    <div className="space-y-6">
   323→      {/* Stats Cards */}
   324→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   325→        <Card className="border-emerald-500/20">
   326→          <CardContent className="p-4">
   327→            <div className="flex items-center gap-3">
   328→              <div className="flex size-9 items-center justify-center rounded-lg bg-emerald-500/10">
   329→                <Mail className="size-4 text-emerald-600" />
   330→              </div>
   331→              <div>
   332→                <p className="text-xs text-muted-foreground">Total Sent</p>
   333→                <p className="text-xl font-bold">{isLoadingStats ? '...' : stats.sent}</p>
   334→              </div>
   335→            </div>
   336→          </CardContent>
   337→        </Card>
   338→        <Card className="border-green-500/20">
   339→          <CardContent className="p-4">
   340→            <div className="flex items-center gap-3">
   341→              <div className="flex size-9 items-center justify-center rounded-lg bg-green-500/10">
   342→                <CheckCircle2 className="size-4 text-green-600" />
   343→              </div>
   344→              <div>
   345→                <p className="text-xs text-muted-foreground">Delivered</p>
   346→                <p className="text-xl font-bold">{isLoadingStats ? '...' : stats.recentSent}</p>
   347→              </div>
   348→            </div>
   349→          </CardContent>
   350→        </Card>
   351→        <Card className="border-amber-500/20">
   352→          <CardContent className="p-4">
   353→            <div className="flex items-center gap-3">
   354→              <div className="flex size-9 items-center justify-center rounded-lg bg-amber-500/10">
   355→                <Clock className="size-4 text-amber-600" />
   356→              </div>
   357→              <div>
   358→                <p className="text-xs text-muted-foreground">Pending</p>
   359→                <p className="text-xl font-bold">{isLoadingStats ? '...' : stats.pending}</p>
   360→              </div>
   361→            </div>
   362→          </CardContent>
   363→        </Card>
   364→        <Card className="border-red-500/20">
   365→          <CardContent className="p-4">
   366→            <div className="flex items-center gap-3">
   367→              <div className="flex size-9 items-center justify-center rounded-lg bg-red-500/10">
   368→                <XCircle className="size-4 text-red-600" />
   369→              </div>
   370→              <div>
   371→                <p className="text-xs text-muted-foreground">Failed</p>
   372→                <p className="text-xl font-bold">{isLoadingStats ? '...' : stats.failed}</p>
   373→              </div>
   374→            </div>
   375→          </CardContent>
   376→        </Card>
   377→      </div>
   378→
   379→      {/* Sub-tab Navigation */}
   380→      <div className="flex gap-1.5 border-b border-border pb-0">
   381→        {[
   382→          { id: 'compose', label: 'Compose', icon: Send },
   383→          { id: 'broadcast', label: 'Broadcast', icon: Radio },
   384→          { id: 'history', label: 'History', icon: Clock },
   385→        ].map((tab) => (
   386→          <button
   387→            key={tab.id}
   388→            onClick={() => setSubTab(tab.id as any)}
   389→            className={`inline-flex items-center gap-1.5 px-4 py-2 text-sm font-medium border-b-2 transition-colors ${
   390→              subTab === tab.id
   391→                ? 'border-emerald-500 text-emerald-600'
   392→                : 'border-transparent text-muted-foreground hover:text-foreground'
   393→            }`}
   394→          >
   395→            <tab.icon className="size-3.5" />
   396→            {tab.label}
   397→          </button>
   398→        ))}
   399→      </div>
   400→
   401→      {/* COMPOSE TAB */}
   402→      {subTab === 'compose' && (
   403→        <Card>
   404→          <CardHeader>
   405→            <div className="flex items-center gap-2">
   406→              <Send className="size-5 text-emerald-600" />
   407→              <CardTitle className="text-lg">Compose Email</CardTitle>
   408→            </div>
   409→            <CardDescription>
   410→              Send an email to a specific user
   411→            </CardDescription>
   412→          </CardHeader>
   413→          <CardContent className="space-y-5">
   414→            {/* Recipient Search */}
   415→            <div className="space-y-2">
   416→              <Label>Recipient</Label>
   417→              {selectedRecipient ? (
   418→                <div className="flex items-center gap-3 p-3 rounded-lg border border-emerald-500/30 bg-emerald-50/50 dark:bg-emerald-950/20">
   419→                  <div className="flex size-8 items-center justify-center rounded-full bg-emerald-500/20 text-emerald-600 text-sm font-bold">
   420→                    {selectedRecipient.firstName?.[0]}{selectedRecipient.lastName?.[0]}
   421→                  </div>
   422→                  <div className="flex-1">
   423→                    <p className="text-sm font-medium">{selectedRecipient.firstName} {selectedRecipient.lastName}</p>
   424→                    <p className="text-xs text-muted-foreground">{selectedRecipient.email} · {selectedRecipient.role}</p>
   425→                  </div>
   426→                  <Button
   427→                    variant="ghost"
   428→                    size="sm"
   429→                    className="text-xs text-red-500 hover:text-red-700"
   430→                    onClick={() => { setSelectedRecipient(null); setRecipientSearch('') }}
   431→                  >
   432→                    Remove
   433→                  </Button>
   434→                </div>
   435→              ) : (
   436→                <div className="relative">
   437→                  <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   438→                  <Input
   439→                    placeholder="Search by name or email..."
   440→                    className="pl-9"
   441→                    value={recipientSearch}
   442→                    onChange={(e) => setRecipientSearch(e.target.value)}
   443→                  />
   444→                  {recipientResults.length > 0 && (
   445→                    <div className="absolute top-full left-0 right-0 z-50 mt-1 bg-popover border rounded-lg shadow-lg max-h-60 overflow-y-auto">
   446→                      {recipientResults.map((u: any) => (
   447→                        <button
   448→                          key={u.id}
   449→                          className="w-full text-left px-3 py-2 hover:bg-muted/50 flex items-center gap-3"
   450→                          onClick={() => {
   451→                            setSelectedRecipient(u)
   452→                            setRecipientResults([])
   453→                            setRecipientSearch('')
   454→                          }}
   455→                        >
   456→                          <div className="flex size-7 items-center justify-center rounded-full bg-emerald-500/20 text-emerald-600 text-xs font-bold">
   457→                            {u.firstName?.[0]}{u.lastName?.[0]}
   458→                          </div>
   459→                          <div>
   460→                            <p className="text-sm font-medium">{u.firstName} {u.lastName}</p>
   461→                            <p className="text-xs text-muted-foreground">{u.email}</p>
   462→                          </div>
   463→                        </button>
   464→                      ))}
   465→                    </div>
   466→                  )}
   467→                </div>
   468→              )}
   469→            </div>
   470→
   471→            {/* Subject */}
   472→            <div className="space-y-2">
   473→              <Label>Subject</Label>
   474→              <Input
   475→                placeholder="Email subject line"
   476→                value={composeSubject}
   477→                onChange={(e) => setComposeSubject(e.target.value)}
   478→              />
   479→            </div>
   480→
   481→            {/* Template */}
   482→            <div className="space-y-2">
   483→              <Label>Email Template</Label>
   484→              <Select value={composeTemplate} onValueChange={setComposeTemplate}>
   485→                <SelectTrigger>
   486→                  <SelectValue placeholder="Select template" />
   487→                </SelectTrigger>
   488→                <SelectContent>
   489→                  {TEMPLATE_OPTIONS.map((opt) => (
   490→                    <SelectItem key={opt.value} value={opt.value}>
   491→                      {opt.label}
   492→                    </SelectItem>
   493→                  ))}
   494→                </SelectContent>
   495→              </Select>
   496→            </div>
   497→
   498→            {/* Message */}
   499→            <div className="space-y-2">
   500→              <Label>Message</Label>
   501→              <Textarea
   502→                placeholder="Type your message here..."
   503→                className="min-h-[150px]"
   504→                value={composeMessage}
   505→                onChange={(e) => setComposeMessage(e.target.value)}
   506→              />
   507→            </div>
   508→
   509→            {/* CTA (optional) */}
   510→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   511→              <div className="space-y-2">
   512→                <Label className="text-muted-foreground">CTA URL (optional)</Label>
   513→                <Input
   514→                  placeholder="https://..."
   515→                  value={composeCtaUrl}
   516→                  onChange={(e) => setComposeCtaUrl(e.target.value)}
   517→                />
   518→              </div>
   519→              <div className="space-y-2">
   520→                <Label className="text-muted-foreground">CTA Label (optional)</Label>
   521→                <Input
   522→                  placeholder="Go to Dashboard"
   523→                  value={composeCtaLabel}
   524→                  onChange={(e) => setComposeCtaLabel(e.target.value)}
   525→                />
   526→              </div>
   527→            </div>
   528→
   529→            {/* Send Button */}
   530→            <Button
   531→              className="w-full bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 text-white"
   532→              onClick={handleSendEmail}
   533→              disabled={isSending || !selectedRecipient || !composeSubject}
   534→            >
   535→              {isSending ? (
   536→                <>
   537→                  <Loader2 className="size-4 animate-spin mr-2" />
   538→                  Sending...
   539→                </>
   540→              ) : (
   541→                <>
   542→                  <Send className="size-4 mr-2" />
   543→                  Send Email
   544→                </>
   545→              )}
   546→            </Button>
   547→          </CardContent>
   548→        </Card>
   549→      )}
   550→
   551→      {/* BROADCAST TAB */}
   552→      {subTab === 'broadcast' && (
   553→        <Card>
   554→          <CardHeader>
   555→            <div className="flex items-center gap-2">
   556→              <Radio className="size-5 text-teal-600" />
   557→              <CardTitle className="text-lg">Broadcast Email</CardTitle>
   558→            </div>
   559→            <CardDescription>
   560→              Send the same email to all users, or filter by role. Max 100 recipients per broadcast.
   561→            </CardDescription>
   562→          </CardHeader>
   563→          <CardContent className="space-y-5">
   564→            {/* Role Filter */}
   565→            <div className="space-y-2">
   566→              <Label>Send to</Label>
   567→              <Select value={broadcastRole} onValueChange={setBroadcastRole}>
   568→                <SelectTrigger>
   569→                  <SelectValue placeholder="Select recipients" />
   570→                </SelectTrigger>
   571→                <SelectContent>
   572→                  <SelectItem value="ALL">All Active Users</SelectItem>
   573→                  <SelectItem value="NURSE">All Nurses</SelectItem>
