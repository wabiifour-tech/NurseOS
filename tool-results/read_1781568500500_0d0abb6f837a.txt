   134→   134→
   135→   135→  // History

   138→   138→  const [historyPage, setHistoryPage] = React.useState(1)
   139→   139→  const [historyTotal, setHistoryTotal] = React.useState(0)
   140→   140→  const [statusFilter, setStatusFilter] = React.useState('ALL')
   141→   141→  const [searchQuery, setSearchQuery] = React.useState('')
   142→   142→
   143→   143→  // Compose email
   144→   144→  const [isSending, setIsSending] = React.useState(false)
   145→   145→  const [recipientSearch, setRecipientSearch] = React.useState('')

   154→   154→  // Broadcast

   174→   174→    } catch (error) {

   176→   176→    } finally {
   177→   177→      setIsLoadingStats(false)
   178→   178→    }
   179→   179→  }, [token])
   180→   180→
   181→   181→  /* ─── Fetch History ─── */

   199→   199→      setIsLoadingEmails(false)
   200→   200→    }

   243→   243→        method: 'POST',
   244→   244→        headers: getHeaders(),
   245→   245→        body: JSON.stringify({
   246→   246→          recipientId: selectedRecipient.id,
   247→   247→          subject: composeSubject,
   248→   248→          templateId: composeTemplate,
   249→   249→          message: composeMessage,
   250→   250→          ctaUrl: composeCtaUrl || undefined,
   251→   251→          ctaLabel: composeCtaLabel || undefined,
   252→   252→        }),
   253→   253→      })

   257→   257→        toast.error(data.error || 'Failed to send email')
   258→   258→        return
   259→   259→      }
   260→   260→

   264→   264→      setRecipientSearch('')

   268→   268→      setComposeCtaLabel('')
   269→   269→      fetchStats()
   270→   270→    } catch (error) {
   271→   271→      console.error('Send email error:', error)
   272→   272→      toast.error('Failed to send email')
   273→   273→    } finally {
   274→   274→      setIsSending(false)
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  /* ─── Broadcast Email ─── */
   279→   279→  const handleBroadcast = async () => {

   281→   281→      toast.error('Please enter a subject')
   282→   282→      return
   283→   283→    }

   287→   287→        subject: broadcastSubject,
   288→   288→        templateId: broadcastTemplate,
   289→   289→        message: broadcastMessage,
   290→   290→      }

   296→   296→      const res = await fetch('/api/email/broadcast', {
   297→   297→        method: 'POST',
   298→   298→        headers: getHeaders(),
   299→   299→        body: JSON.stringify(body),
   300→   300→      })
   301→   301→      const data = await res.json()
   302→   302→
   303→   303→      if (!res.ok) {
   304→   304→        toast.error(data.error || 'Failed to broadcast')
   305→   305→        return
   306→   306→      }
   307→   307→
   308→   308→      toast.success(data.message || `Broadcast sent to ${data.totalSent} users`)
   309→   309→      setBroadcastSubject('')
   310→   310→      setBroadcastMessage('')

   313→   313→      console.error('Broadcast error:', error)
   314→   314→      toast.error('Failed to broadcast email')

   328→   328→              <div className="flex size-9 items-center justify-center rounded-lg bg-emerald-500/10">
   329→   329→                <Mail className="size-4 text-emerald-600" />
   330→   330→              </div>

   347→   347→              </div>
   348→   348→            </div>
   349→   349→          </CardContent>
   350→   350→        </Card>
   351→   351→        <Card className="border-amber-500/20">
